package registrygenesis_test

// Applied by tests/joined/run.py as a Go overlay on the pinned oracle; never committed there.
// Runs the production B1 genesis path (q3format.NewHistory -> registrygenesis.GenerateB1) over the
// fixed one-member root authority of the released corpus ("root-1"), for the golden dev deployment
// (network 3, EVM partition 7, chain 31337) via bridgeprofile.NewDeployment.

import (
	"encoding/hex"
	"encoding/json"
	"os"
	"testing"

	"github.com/ethereum/go-ethereum/common"
	"github.com/stretchr/testify/require"
	"github.com/unicitynetwork/bft-core/b1registry"
	"github.com/unicitynetwork/bft-core/bridgeprofile"
	"github.com/unicitynetwork/bft-core/b1state"
	"github.com/unicitynetwork/bft-core/internal/quorumweight"
	"github.com/unicitynetwork/bft-core/q3format"
	"github.com/unicitynetwork/bft-core/registrygenesis"
	"github.com/unicitynetwork/bft-go-base/types"
)

func TestNBPPR6NativeGenesis(t *testing.T) {
	out := os.Getenv("NBP_PR6_NATIVE_OUT")
	if out == "" {
		t.Skip("NBP_PR6_NATIVE_OUT not set")
	}
	d, err := bridgeprofile.NewDeployment()
	require.NoError(t, err)
	// The B1 registry genesis is built from the root's V1 genesis trust base: the corpus authority's one
	// member (root-1), epoch 1 from round 0, signed by it (the bridge trust input is its V2 history, d.TB).
	m := d.TB.RootNodes[0]
	tb, err := quorumweight.NewTrustBase(d.TB.NetworkID, []*types.NodeInfo{{NodeID: m.NodeID, SigKey: m.SigKey, Stake: m.Stake}},
		types.WithEpoch(1), types.WithEpochStart(0))
	require.NoError(t, err)
	require.NoError(t, tb.Sign(d.Auth.NodeID, d.Auth.Signer))
	h, err := q3format.NewHistory(tb)
	require.NoError(t, err)

	const b = 2 // B, the epochs one block may install (the registry's measured cap)
	p := b1state.Profile{Network: 3, RootGenesisID: h.Genesis(), ExecutionChainID: 31337,
		RuntimeHash: [32]byte(common.HexToHash(b1registry.CodeHashHex)), CompilerHash: b1registry.CompilerHash(),
		MaxEpochsPerUpdate: b, DeltaEV: 16, DeltaHold: 17, RestGas: b1registry.MinRestGas(b),
		CompanionBytes: 1 << 20, OtherCompanionBytes: 65536, OrdinaryCapacity: 7000000, GenesisUCTime: 1000}
	p.SystemGas, err = p.RequiredSystemGas()
	require.NoError(t, err)
	p.MaxGas = p.SystemGas + p.OrdinaryCapacity
	profileHash, err := p.Hash()
	require.NoError(t, err)

	// The oracle fixture names the parameter "chainId"; production genesis requires "chain_id".
	confValue := *d.EVMPDR
	confValue.PartitionParams = map[string]string{registrygenesis.ChainIDParam: "31337"}
	conf := &confValue
	g, err := registrygenesis.GenerateB1(conf, p, h, registrygenesis.EVMParams{GasLimit: p.MaxGas, BaseFee: registrygenesis.DefaultEVMParams.BaseFee})
	require.NoError(t, err)

	words := map[string]string{}
	for k, v := range g.B1Words() {
		words[k.Hex()] = v.Hex()
	}
	doc := map[string]any{
		"network": p.Network, "executionChainId": p.ExecutionChainID, "maxEpochsPerUpdate": p.MaxEpochsPerUpdate,
		"rootGenesisId": hex.EncodeToString(p.RootGenesisID[:]), "profileHash": hex.EncodeToString(profileHash[:]),
		"runtimeHash": hex.EncodeToString(p.RuntimeHash[:]), "systemGas": p.SystemGas, "maxGas": p.MaxGas,
		"ordinaryCapacity": p.OrdinaryCapacity, "restGas": p.RestGas,
		"evmGenesisHash": g.EVMGenesisHash().Hex(), "genesisCommitment": g.GenesisCommitment().Hex(), "shardConfHash": g.FullShardConfHash().Hex(),
		"registryWords": words, "genesis": json.RawMessage(g.GenesisJSON()),
	}
	enc, err := json.MarshalIndent(doc, "", "  ")
	require.NoError(t, err)
	require.NoError(t, os.WriteFile(out, append(enc, '\n'), 0o644))
}
